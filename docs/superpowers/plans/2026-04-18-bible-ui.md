# Bible UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Créer le package frontend `@buck/bible-ui` (SPA Vite/React + nginx) qui consomme le serveur `bible-mcp` via JSON-RPC, déployé en 3ème service Docker derrière Caddy basicauth privé sur `bible.buck.romain-ecarnot.com`.

**Architecture:** 3 services Docker indépendants (`buck-app`, `bible-ui`, `bible-mcp`). Le SPA bible-ui est servi par un nginx alpine qui proxifie aussi `/mcp` vers `bible-mcp:7801` (réseau Docker interne). Same-origin partout, zéro CORS browser. En dev, Vite proxy `/mcp` vers `localhost:7801`.

**Tech Stack:** React 19, Vite 6, TanStack Router file-based, TanStack Query 5, Tailwind 4.2, shadcn/ui (preset b1Gdz9c4A, baseColor stone), Sigma + graphology, Lucide, Inter+JetBrainsMono+Georgia, nginx alpine, Caddy basicauth. Design erom v2 amber dark-first repris depuis `@buck/web`.

**Spec:** [docs/superpowers/specs/2026-04-18-bible-ui-design.md](../specs/2026-04-18-bible-ui-design.md)

```ts
import type { JsonRpcResponse, McpTool } from '@/types/mcp';

let requestId = 0;

export async function callTool(name: string, params: Record<string, unknown> = {}): Promise<unknown> {
  const res = await fetch('/mcp', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      jsonrpc: '2.0',
      id: ++requestId,
      method: 'tools/call',
      params: { name, arguments: params },
    }),
  });
  if (!res.ok) throw new Error(`HTTP ${res.status}: ${res.statusText}`);
  const json = (await res.json()) as JsonRpcResponse;
  if (json.error) throw new Error(json.error.message);
  const result = json.result;
  if (!result?.content?.length) return null;
  if (result.isError) throw new Error(result.content[0].text);
  return JSON.parse(result.content[0].text);
}

export async function listTools(): Promise<McpTool[]> {
  const res = await fetch('/mcp', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ jsonrpc: '2.0', id: ++requestId, method: 'tools/list' }),
  });
  if (!res.ok) throw new Error(`HTTP ${res.status}: ${res.statusText}`);
  const json = (await res.json()) as JsonRpcResponse;
  if (json.error) throw new Error(json.error.message);
  return json.result?.tools ?? [];
}
```

```ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { callTool } from '@/api/mcp-client';

export function useMcpQuery<T = unknown>(
  toolName: string,
  params?: Record<string, unknown>,
  options?: { enabled?: boolean }
) {
  return useQuery<T>({
    queryKey: [toolName, params],
    queryFn: () => callTool(toolName, params) as Promise<T>,
    ...options,
  });
}

export function useMcpMutation(toolName: string, invalidateKeys?: string[]) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (params: Record<string, unknown>) => callTool(toolName, params),
    onSuccess: () => {
      invalidateKeys?.forEach((k) => queryClient.invalidateQueries({ queryKey: [k] }));
    },
  });
}
```

```ts
import { useMcpQuery } from './use-mcp';
import type { AnyEntity, Interaction } from '@/types/entities';

export interface GraphData {
  nodes: Array<{ id: string; label: string; type: string; color: string }>;
  edges: Array<{ id: string; source: string; target: string; label?: string }>;
}

const COLORS: Record<string, string> = {
  character: '#60a5fa', location: '#34d399', event: '#fbbf24',
  note: '#22d3ee', research: '#a78bfa', 'world-rule': '#f472b6',
};

export function useGraph(): { data: GraphData; isLoading: boolean } {
  const characters = useMcpQuery<AnyEntity[]>('bible.characters.list');
  const locations = useMcpQuery<AnyEntity[]>('bible.locations.list');
  const events = useMcpQuery<AnyEntity[]>('bible.events.list');
  const interactions = useMcpQuery<Interaction[]>('bible.interactions.list');

  const isLoading = characters.isLoading || locations.isLoading || events.isLoading || interactions.isLoading;

  const nodes = [...(characters.data ?? []), ...(locations.data ?? []), ...(events.data ?? [])].map((e) => ({
    id: e.id, label: e.name, type: e.type, color: COLORS[e.type] ?? '#6b7280',
  }));

  const edges: GraphData['edges'] = [];
  (interactions.data ?? []).forEach((i) => {
    if (i.participants.length >= 2) {
      for (let k = 0; k < i.participants.length - 1; k++) {
        edges.push({ id: `${i.id}-${k}`, source: i.participants[k], target: i.participants[k + 1], label: i.name });
      }
    }
  });

  return { data: { nodes, edges }, isLoading };
}
```

```tsx
function LoadGraph({ data }: { data: GraphData }) {
  const loadGraph = useLoadGraph();
  useEffect(() => {
    const g = new Graph();
    data.nodes.forEach((n) => g.addNode(n.id, { label: n.label, color: n.color, size: 8, x: Math.random(), y: Math.random() }));
    data.edges.forEach((e) => {

        g.addEdgeWithKey(e.id, e.source, e.target, { label: e.label, color: '#3f3f46' });
      }
    });
    loadGraph(g);
  }, [data, loadGraph]);
  return null;
}
```
