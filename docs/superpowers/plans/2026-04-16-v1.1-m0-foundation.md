# Buck Writer v1.1 — Milestone M0 « Foundation » Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Livrer un socle fonctionnel de Buck Writer : un utilisateur whitelisté peut se logger via magic link email et atterrir sur une page authentifiée vide. Aucune UI chat. Aucun call OpenAI. Aucune Bible. Juste le squelette end-to-end qui prouve que tous les rails fonctionnent (repo + monorepo + Docker + DB + auth + CSP + CSRF + WebDAV token).

**Architecture:** Monorepo pnpm à 3 packages (`@buck/shared` types, `@buck/api` backend Hono, `@buck/web` frontend Vite + React + shadcn preset `b1Gdz9c4A`). SQLite via Drizzle ORM. Auth magic link via Resend. Container Docker multi-stage avec entrypoint qui applique les migrations. Branché sur le réseau Docker externe `caddy-public` du Caddy Trinity existant.

**Tech Stack:** Node 20+, pnpm 9+, TypeScript 5.6+, Hono 4+, Vite 5+, React 19, Drizzle ORM + better-sqlite3, Zod 4, TanStack Router, Tailwind + shadcn/ui (preset `b1Gdz9c4A`), jose (JWT), resend, Vitest, Playwright, Docker + Docker Compose, Caddy.

**Scope M0 — in :**
- Repo skeleton + pnpm monorepo + tooling (prettier, eslint, tsconfig)
- Packages `shared` / `api` / `web` scaffolded
- SQLite schema complet (toutes les tables, même celles utilisées en M1-M5)
- Migration programmatique + seed idempotent
- Auth magic link complet : request + callback + logout
- Middlewares : CSRF double-submit, security headers (CSP/HSTS/etc), rate limit, auth guard
- JWT session cookie httpOnly
- WebDAV token endpoint (settings partiel) — réservé, le handler WebDAV complet arrive en M3
- Path-safe utility
- Web : page `/login` + callback + root layout guard → page `/` vide authentifiée
- Dockerfile multi-stage + docker-compose.yml + entrypoint.sh
- 1 scénario e2e Playwright : `auth-happy-path`
- README-DEV.md documentant le workflow local

**Scope M0 — out (pour les milestones suivants) :**
- Sessions CRUD, Chat, OpenAI streaming → M1
- Métriques, cost guard, alertes → M2
- Attachments, handler WebDAV complet → M3
- Bible MCP → M4
- Frontend chat/metrics/settings UI complet, CI/CD, deploy → M5

**Spec de référence :** `docs/superpowers/specs/2026-04-16-buck-writer-v1-1-foundation-design.md`

- [ ] **Step 9: Write `packages/shared/src/pricing/models.ts`**

```typescript
export const PRICING = {
  'gpt-5.4': { input: 2.5, output: 15.0 },
  'gpt-5.4-mini': { input: 0.75, output: 4.5 },
  'gpt-5.4-pro': { input: 5.0, output: 30.0 },
  'gpt-5.4-nano': { input: 0.15, output: 0.6 },
  'gpt-realtime-1.5': {
    text_input: 5.0,
    text_output: 20.0,
    audio_input: 100.0,
    audio_output: 200.0,
  },
} as const;

export type TextModel = 'gpt-5.4' | 'gpt-5.4-mini' | 'gpt-5.4-pro' | 'gpt-5.4-nano';
export type RealtimeModel = 'gpt-realtime-1.5';
export type Model = TextModel | RealtimeModel;

export function costOf(model: string, inputTokens: number, outputTokens: number): number {
  const p = (PRICING as Record<string, { input: number; output: number } | undefined>)[model];
  if (!p || !('input' in p)) return 0;
  return (inputTokens * p.input + outputTokens * p.output) / 1_000_000;
}
```

```typescript
import { HttpError } from './utils/http-error.js';

export function buildApp() {
  const app = new Hono();

  app.onError((err, c) => {
    if (err instanceof HttpError) {
      return c.json(err.toJSON(), err.status as 400);
    }
    console.error('[api] unhandled error', err);
    return c.json({ error: { code: 'internal', message: 'internal server error' } }, 500);
  });

  app.notFound((c) => c.json({ error: { code: 'not_found', message: 'route not found' } }, 404));

  return app;
}
```

```typescript
export interface SeedOptions {
  databaseUrl: string;
  allowedEmails: string[];
  mcpBibleUrl: string;
}

export function runSeed(opts: SeedOptions): void {
  const filePath = opts.databaseUrl.replace(/^file:/, '');
  const sqlite = new Database(filePath);
  sqlite.pragma('foreign_keys = ON');
  const now = Date.now();

  // 1. MCP Bible core
  const insertMcp = sqlite.prepare(`
    INSERT INTO mcp_servers (id, name, core, enabled, transport, config_json, created_at)
    VALUES (@id, 'bible', 1, 1, 'http-streamable', @config, @createdAt)
    ON CONFLICT(name) DO UPDATE SET
      config_json = excluded.config_json,
      core = 1
  `);

    config: JSON.stringify({ url: `${opts.mcpBibleUrl}/mcp` }),
    createdAt: now,
  });

  // 2. Users whitelistés + user_settings
  const insertUser = sqlite.prepare(`
    INSERT INTO users (id, email, created_at)
    VALUES (@id, @email, @createdAt)
    ON CONFLICT(email) DO NOTHING
  `);
  const insertSettings = sqlite.prepare(`
    INSERT INTO user_settings (user_id)
    VALUES (@userId)
    ON CONFLICT(user_id) DO NOTHING
  `);

  for (const email of opts.allowedEmails) {
    const id = newId();
    insertUser.run({ id, email, createdAt: now });
    const user = selectUser.get(email) as { id: string } | undefined;
    if (user) insertSettings.run({ userId: user.id });
  }

  sqlite.close();
}
```

```typescript
import path from 'node:path';
import fs from 'node:fs/promises';
import { HttpError } from './http-error.js';

export async function assertSafePath(rootDir: string, relative: string): Promise<string> {
  const root = path.resolve(rootDir);
  const joined = path.resolve(root, relative);
  let real = joined;
  try {
    real = await fs.realpath(joined);
  } catch {
    // path doesn't exist yet (e.g. PUT a new file) — use resolved joined path
  }
  if (real !== root && !real.startsWith(root + path.sep)) {
    throw new HttpError(403, 'forbidden_path', 'path outside workspace');
  }
  return real;
}

export function createPathSafe(rootDir: string) {
  return (relative: string) => assertSafePath(rootDir, relative);
}
```

```typescript
import type { MiddlewareHandler } from 'hono';

interface Bucket {
  count: number;
  resetAt: number;
}

export interface RateLimitOptions {
  windowMs: number;
  max: number;
  keyBy: (c: Parameters<MiddlewareHandler>[0]) => string;
}

export function createRateLimiter(opts: RateLimitOptions): MiddlewareHandler {
  const buckets = new Map<string, Bucket>();
  return async (c, next) => {
    const key = opts.keyBy(c);
    const now = Date.now();
    const bucket = buckets.get(key);
    if (!bucket || bucket.resetAt <= now) {
      buckets.set(key, { count: 1, resetAt: now + opts.windowMs });
      return next();
    }
    if (bucket.count >= opts.max) {
      return c.json(
        { error: { code: 'rate_limited', message: 'too many requests' } },
        429,
        { 'Retry-After': String(Math.ceil((bucket.resetAt - now) / 1000)) },
      );
    }
    bucket.count += 1;
    return next();
  };
}

export const ipKey = (c: Parameters<MiddlewareHandler>[0]): string => {
  const fwd = c.req.header('x-forwarded-for');
  if (fwd) return fwd.split(',')[0]!.trim();
  return c.req.header('cf-connecting-ip') ?? 'unknown';
};
```

```typescript
import type { MiddlewareHandler } from 'hono';
import crypto from 'node:crypto';

export const CSRF_COOKIE = 'buck_csrf';
export const CSRF_HEADER = 'x-csrf-token';
const SAFE_METHODS = new Set(['GET', 'HEAD', 'OPTIONS']);

function randomToken(): string {
  return crypto.randomBytes(24).toString('hex');
}

function parseCookies(raw: string | null | undefined): Record<string, string> {
  if (!raw) return {};
  return Object.fromEntries(
    raw.split(';').map((p) => {
      const [k, ...v] = p.trim().split('=');
      return [k ?? '', decodeURIComponent((v.join('=') ?? '').trim())];
    }),
  );
}

export function csrfMiddleware(): MiddlewareHandler {
  return async (c, next) => {
    const cookies = parseCookies(c.req.header('cookie'));
    const method = c.req.method.toUpperCase();

    if (SAFE_METHODS.has(method)) {
      if (!cookies[CSRF_COOKIE]) {
        const fresh = randomToken();
        c.header(
          'Set-Cookie',
          `${CSRF_COOKIE}=${fresh}; Path=/; SameSite=Lax; Secure; Max-Age=2592000`,
        );
      }
      return next();
    }

    const cookie = cookies[CSRF_COOKIE];
    const header = c.req.header(CSRF_HEADER);
    if (!cookie || !header) {
      return c.json(
        { error: { code: 'csrf_missing', message: 'CSRF token missing' } },
        403,
      );
    }
    if (cookie !== header) {
      return c.json(
        { error: { code: 'csrf_mismatch', message: 'CSRF token mismatch' } },
        403,
      );
    }
    return next();
  };
}
```

```typescript
import { SignJWT, jwtVerify } from 'jose';

export interface JwtOptions {
  secret: string;
  issuer: string;
  audience: string;
}

export interface JwtPayload {
  sub: string;
  scope?: 'app' | 'webdav';
  [key: string]: unknown;
}

export function createJwtService(opts: JwtOptions) {
  const key = new TextEncoder().encode(opts.secret);

  return {
    async sign(payload: JwtPayload, expiresIn: string): Promise<string> {
      return new SignJWT(payload as Record<string, unknown>)
        .setProtectedHeader({ alg: 'HS256' })
        .setIssuedAt()
        .setIssuer(opts.issuer)
        .setAudience(opts.audience)
        .setExpirationTime(expiresIn)
        .sign(key);
    },
    async verify(token: string): Promise<JwtPayload> {
      const { payload } = await jwtVerify(token, key, {
        issuer: opts.issuer,
        audience: opts.audience,
      });
      return payload as JwtPayload;
    },
  };
}

export type JwtService = ReturnType<typeof createJwtService>;
```

```typescript
import { Hono } from 'hono';
import { AuthRequestInput, newId } from '@buck/shared';
import type { DbHandles } from '../db/client.js';
import type { EmailService } from '../services/email.js';
import type { JwtService } from '../services/jwt.js';
import { authTokens, users } from '../db/schema.js';
import { sha256Hex, randomTokenHex } from '../utils/crypto.js';
import { eq } from 'drizzle-orm';

export interface AuthRoutesDeps {
  db: DbHandles;
  email: EmailService;
  jwt: JwtService;
  allowedEmails: string[];
  publicBaseUrl: string;
  nowMs?: () => number;
}

const MAGIC_LINK_TTL_MS = 15 * 60 * 1000;

export function createAuthRoutes(deps: AuthRoutesDeps): Hono {
  const now = deps.nowMs ?? Date.now;
  const whitelist = new Set(deps.allowedEmails.map((e) => e.toLowerCase()));
  const app = new Hono();

  app.post('/request', async (c) => {
    const raw = await c.req.json().catch(() => ({}));
    const parsed = AuthRequestInput.safeParse(raw);
    if (!parsed.success) {
      return c.json(
        { error: { code: 'invalid_input', message: 'invalid email' } },
        400,
      );
    }
    const email = parsed.data.email.toLowerCase();
    const ts = now();

    // Constant-time-ish response: always 200 to avoid enumeration.
    if (!whitelist.has(email)) {
      // simulate delay ~constant
      await new Promise((r) => setTimeout(r, 400));
      return c.json({ sent: true });
    }

    // Ensure user exists (seed already inserts whitelisted users)
    const existing = await deps.db.db.select().from(users).where(eq(users.email, email)).get();
    if (!existing) {
      return c.json({ sent: true }); // defensive
    }

    const token = randomTokenHex(32);
    const tokenHash = sha256Hex(token);
    await deps.db.db.insert(authTokens).values({
      id: newId(),
      userId: existing.id,
      tokenHash,
      expiresAt: ts + MAGIC_LINK_TTL_MS,
      createdAt: ts,
    });

    const magicUrl = `${deps.publicBaseUrl}/api/auth/callback?token=${token}`;
    await deps.email.sendMagicLink({ to: email, magicUrl });

    return c.json({ sent: true });
  });

  return app;
}
```

```typescript
import type { MiddlewareHandler } from 'hono';
import { eq, and, gte } from 'drizzle-orm';
import type { DbHandles } from '../db/client.js';
import type { JwtService } from '../services/jwt.js';
import { sessionsAuth } from '../db/schema.js';
import { sha256Hex } from '../utils/crypto.js';

export interface AuthGuardDeps {
  db: DbHandles;
  jwt: JwtService;
  nowMs?: () => number;
}

function parseCookies(raw: string | null | undefined): Record<string, string> {
  if (!raw) return {};
  return Object.fromEntries(
    raw.split(';').map((p) => {
      const [k, ...v] = p.trim().split('=');
      return [k ?? '', decodeURIComponent((v.join('=') ?? '').trim())];
    }),
  );
}

export function authGuard(deps: AuthGuardDeps): MiddlewareHandler {
  const now = deps.nowMs ?? Date.now;
  return async (c, next) => {
    const cookies = parseCookies(c.req.header('cookie'));
    const raw = cookies['buck_session'];
    if (!raw) {
      return c.json({ error: { code: 'unauthenticated', message: 'no session' } }, 401);
    }
    try {
      const payload = await deps.jwt.verify(raw);
      const hash = sha256Hex(raw);
      const session = await deps.db.db
        .select()
        .from(sessionsAuth)
        .where(and(eq(sessionsAuth.tokenHash, hash), gte(sessionsAuth.expiresAt, now())))
        .get();
      if (!session || session.userId !== payload.sub) {
        return c.json({ error: { code: 'unauthenticated', message: 'session revoked' } }, 401);
      }
      c.set('userId', payload.sub);
      await next();
      return;
    } catch {
      return c.json({ error: { code: 'unauthenticated', message: 'invalid token' } }, 401);
    }
  };
}
```

```typescript
import { CSRF_HEADER, readCsrfCookie } from './csrf';

export class ApiError extends Error {
  constructor(
    public status: number,
    public code: string,
    message: string,
  ) {
    super(message);
  }
}

export async function apiFetch<T>(
  path: string,
  init: RequestInit & { body?: unknown } = {},
): Promise<T> {
  const headers = new Headers(init.headers);
  if (init.body !== undefined && !headers.has('content-type')) {
    headers.set('content-type', 'application/json');
  }
  const method = (init.method ?? 'GET').toUpperCase();
  if (!['GET', 'HEAD', 'OPTIONS'].includes(method)) {
    headers.set(CSRF_HEADER, readCsrfCookie());
  }
  const res = await fetch(path, {
    ...init,
    headers,
    body: init.body !== undefined && typeof init.body !== 'string' ? JSON.stringify(init.body) : (init.body as BodyInit | undefined),
    credentials: 'include',
  });
  const ct = res.headers.get('content-type') ?? '';
  const json = ct.includes('application/json') ? ((await res.json()) as unknown) : undefined;
  if (!res.ok) {
    const err = (json as { error?: { code?: string; message?: string } } | undefined)?.error ?? {};
    throw new ApiError(res.status, err.code ?? 'unknown', err.message ?? res.statusText);
  }
  return json as T;
}
```

```tsx
      <h1 className="mb-8 text-3xl font-semibold">buck writer</h1>

          onChange={(e) => setEmail(e.target.value)}
          className="rounded-md border border-border bg-input px-3 py-2 text-foreground"
```

```typescript
export function createE2EEmailService(tokenFilePath: string) {
  return {
    async sendMagicLink(params: { to: string; magicUrl: string }): Promise<void> {
      const fs = await import('node:fs/promises');
      const url = new URL(params.magicUrl);
      const rawToken = url.searchParams.get('token') ?? '';
      await fs.writeFile(
        tokenFilePath,
        JSON.stringify({ email: params.to, rawToken }, null, 2),
        'utf8',
      );
    },
  };
}
```
